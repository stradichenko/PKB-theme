{{- /* Citation shortcode. Usage: {{< cite "key" "references.bib" "apa" "false" >}} Stores citation state on .Page.Store (Hugo 0.85+). */ -}} {{- $citationKey := .Get 0 | default "" -}} {{- $bibFile := .Get 1 | default "references.bib" -}} {{- $existingNum := .Page.Store.Get (printf "citation-%s-num" $citationKey) -}} {{- if $existingNum -}} {{- $existingCitation := index (.Page.Store.Get "citations") $citationKey -}} {{ $existingNum }} {{- else -}} {{- $num := .Page.Store.Get "citeCounter" | default 0 -}} {{- $num = add $num 1 -}} {{- .Page.Store.Set "citeCounter" $num -}} {{- .Page.Store.Set (printf "citation-%s-num" $citationKey) $num -}} {{- $style := .Get 2 | default "apa" -}} {{- $showNotes := .Get 3 | default "false" -}} {{- $citationId := printf "citation-%s" $citationKey -}} {{- $lang := site.Language.Lang | default "en" -}} {{ $num }}{{ $num }}. {{- $citations := .Page.Store.Get "citations" | default dict -}} {{- $citations = merge $citations (dict $citationKey (dict "id" $citationId "style" $style "bibFile" $bibFile)) -}} {{- .Page.Store.Set "citations" $citations -}} {{- end -}}
